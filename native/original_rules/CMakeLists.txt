cmake_minimum_required(VERSION 3.29)
project(tm_original_rules LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The native rules bridge requires MSVC x86: configure with -A Win32")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX23_STANDARD_COMPILE_OPTION "/std:c++23preview")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(tm_original_rules SHARED bridge.cpp engine.cpp exports.def)
target_compile_options(tm_original_rules PRIVATE /W4 /WX /EHsc /utf-8)
target_compile_definitions(tm_original_rules PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_options(tm_original_rules PRIVATE /BASE:0x60000000)
